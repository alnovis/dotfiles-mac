  122  
  123          echo "---"
  124          set -l summary "$unpushed unpushed commit(s), $total_files file(s)"
  125          if test $total_add -gt 0
  126              set summary "$summary, "(set_color green)"+$total_add"(set_color normal)
  127          end
  128          if test $total_del -gt 0
  129              set summary "$summary, "(set_color red)"-$total_del"(set_color normal)
  130          end
  131          echo $summary
  132          return 0
  133      end
  134  
  135      # Find merge base
  136      set -l merge_base (git merge-base origin/$base HEAD 2>/dev/null)
  137      if test -z "$merge_base"
  138          set merge_base (git merge-base $base HEAD 2>/dev/null)
  139          if test -z "$merge_base"
  140              echo "Cannot find common ancestor with $base"
  141              return 1
  142          end
  143      end
  144  
  145      set -l ahead (git rev-list --count $merge_base..HEAD)
  146  
  147      echo "Repository: $repo_name ($branch)"
  148      set_color cyan
  149      echo "Base: $base ($ahead commit(s) ahead)"
  150      set_color normal
  151  
  152      if test "$ahead" -eq 0
  153          echo "---"
  154          echo "No commits ahead of $base"
  155          return 0
  156      end
  157  
  158      # Commits
  159      echo ""
  160      set_color green
  161      echo "Commits:"
  162      set_color normal
  163      git log --oneline $color_flag $merge_base..HEAD | while read -l line
  164          echo "  $line"
  165      end
  166  
  167      # Diff stat
  168      set -l stat_flag --stat
  169      if set -q _flag_stat
  170          set stat_flag --stat=$_flag_stat
  171      else if test -n "$COLUMNS"
  172          set stat_flag --stat=$COLUMNS
  173      end
  174  
  175      echo ""
  176      set_color yellow
  177      echo "Files:"
  178      set_color normal
  179      set -l diff_stat (git diff $color_flag $stat_flag $merge_base..HEAD | string collect)
  180      if test -n "$diff_stat"
  181          echo "$diff_stat"
  182      end
  183  
  184      # Summary
  185      set -l nums (git diff --numstat $merge_base..HEAD | string match -rv '^\s*$')
  186      set -l total_add 0
  187      set -l total_del 0
  188      set -l total_files 0
  189  
  190      for line in $nums
  191          set -l parts (string split \t $line)
  192          if test "$parts[1]" != "-"
  193              set total_add (math $total_add + $parts[1])
  194              set total_del (math $total_del + $parts[2])
  195          end
  196          set total_files (math $total_files + 1)
  197      end
  198  
  199      echo "---"
  200      set -l summary "$ahead commit(s), $total_files file(s)"
  201      if test $total_add -gt 0
  202          set summary "$summary, "(set_color green)"+$total_add"(set_color normal)
  203      end
  204      if test $total_del -gt 0
  205          set summary "$summary, "(set_color red)"-$total_del"(set_color normal)
  206      end
  207      echo $summary
  208  end
